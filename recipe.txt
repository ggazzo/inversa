# enum Steps

# };


SET_POWER 3200
SET_VOLUME 70

# Recipe with confirmation steps, timing, step tracking and boil phase
# Set estimated time for the entire recipe (in seconds)
# 7200 seconds = 2 hours
SET_ESTIMATED_TIME 7200

# Start the recipe
START

# Set step 1 - Initial heating
SET_STEP 1
# Set target temperature to 65°C
SET_TEMP 65

# Wait for temperature to reach target
WAIT_TEMP 65

# Wait for confirmation before proceeding
WAIT_CONFIRM "Temperature reached 65°C. Proceed?"

# Set step Mashing
SET_STEP 2
# Start a timer for 30 minutes
WAIT_TIMER 1800
# Wait for confirmation before proceeding
WAIT_CONFIRM "30 minutes passed. Proceed to next step?"

# Set step Mashing
SET_STEP 2
# Set temperature to 75°C
SET_TEMP 75
# Wait for temperature to reach target
WAIT_TEMP 75
# Wait for confirmation before proceeding
WAIT_CONFIRM "Temperature reached 75°C. Proceed?"

# Set step Mashing out
SET_STEP 3

SET_TEMP 80
# Start a timer for 15 minutes
WAIT_TIMER 900
# Wait for confirmation before proceeding
WAIT_CONFIRM "15 minutes passed. Proceed to next step?"

# Set step Sparging
SET_STEP 4

WAIT_CONFIRM "Sparging step. Proceed to next step?"

# Set step 5 - Boil phase
SET_STEP 5
# Set temperature to 100°C for boiling
SET_TEMP 100

# Wait for temperature to reach target
WAIT_TEMP 100

# Start a timer for 60 minutes
WAIT_TIMER 3600

# Wait for confirmation before proceeding
WAIT_CONFIRM "Boil completed. Proceed to next step?"

# Set step Cooling
SET_STEP 6

# Set temperature to 80°C
SET_TEMP 80

# Wait for temperature to reach target
WAIT_TEMP 80


# End the recipe
END
